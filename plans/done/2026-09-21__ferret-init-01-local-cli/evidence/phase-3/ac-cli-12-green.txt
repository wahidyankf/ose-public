# Run the FERRET E2E suite on Linux against the artifact built on the host (the build is reproducible).
# The worktree is mounted read-only and copied into the container, so the host tree is never written to.

  set -eu
  mkdir -p /work/apps/ferret-cli/dist
  cd /host
  tar --exclude=.venv --exclude=__pycache__ -cf - apps/ferret-cli-e2e specs | tar -xf - -C /work
  cp apps/ferret-cli/dist/ferret.pyz /work/apps/ferret-cli/dist/ferret.pyz
  pip install --quiet --disable-pip-version-check pytest pytest-bdd
  cd /work/apps/ferret-cli-e2e
  uname -sr
  python --version
  python -m pytest '"$*"' -p no:cacheprovider -q --color=no
'
